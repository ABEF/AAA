class RPC:

  self.records.append(record)

 helper='0x'+read(pins['nft'],'seaDropMint()')[0][-40:];drop='0x'+read(helper,'seaDrop()')[0][-40:]
 count=int(read(royalty,'assetCount()')[0],16);capacity=int(read(royalty,'assetCapacity()')[0],16)
 assert count==65 and capacity==64
 assets=['0x'+read(royalty,'assets(uint256)',i)[0][-40:]for i in range(65)]
 validate_asset_inventory(count,capacity,assets)
 assert int(read(pins['protocol_policy'],'royaltySyncGas()')[0],16)==1_000_000
 zzz='0x'+read(pins['activation_assets'],'zzz()')[0][-40:]
 assert helper!=ZERO and drop!=ZERO and int(read(helper,'nft()')[0],16)==int(pins['nft'],16)
 assert all(address in case['materialized']for address in assets[1:]+[helper,drop,royalty])
 assert int(read(pins['oracle'],'executionActive()')[0],16)==0
 workload=royalty_workload(case,read,rpc,anchor,royalty,assets,zzz)
 growth=workload['growth'];expected=1000 if growth else 999
 assert int(read(pins['nft'],'mintedCount()')[0],16)==1000
 assert int(read(royalty,'activationCount()')[0],16)==int(read(pins['membership'],'totalEligible()')[0],16)==expected
 assert int(read(royalty,'totalWeight()')[0],16)==(1500 if growth else 1499)
 assert int(read(pins['nft'],'rarityOf(uint256)',case['nft_id'])[0],16)==0
 assert int(read(royalty,'weightOf(uint256)',case['nft_id'])[0],16)==(1 if growth else 0)
 physical=physical_portfolio(case,read,zzz);density=enrollment_storage(alloc,royalty,assets,growth)
 tx={'from':case['sender'],'to':case['destination'],'value':hex(case['value']),'data':case['data'],'gas':hex(CAP)}
 call32=rpc.attempt('eth_call',[tx,anchor]);estimate32=rpc.attempt('eth_estimateGas',[tx,anchor]);trace32=checked_trace(rpc,tx,anchor,omitted)
 estimate=quantity(estimate32['result'])if estimate32['ok']else None
 padded,padding_fits=contribution_padding(estimate)
 exact=None;padded_trace=None
 if call32['ok']:assert call32['result']=='0x'
 if call32['ok']and estimate is not None and 21000<=estimate<=padded<=CAP:
  padded_tx={**tx,'gas':hex(padded)};exact=rpc.attempt('eth_call',[padded_tx,anchor])
  if exact['ok']:
   assert exact['result']=='0x';padded_trace=checked_trace(rpc,padded_tx,anchor,omitted)
 trace32_ok=trace32['ok']and not trace32['result'].get('error')
 padded_trace_ok=bool(padded_trace and padded_trace['ok']and not padded_trace['result'].get('error'))
 admitted=bool(call32['ok']and estimate is not None and 21000<=estimate<=padded<=CAP and exact and exact['ok']and trace32_ok and padded_trace_ok)
 result={'actual_32m_call':call32,'estimate_32m':estimate32,'estimated_gas':estimate,'padded_gas':padded,
  'contribution_profile_cap':CAP,'padding_percent':20,'maximum_estimate_before_padding':26_666_666,
  'exact_padded_call':exact,'trace32_succeeded':trace32_ok,'exact_padded_trace_succeeded':padded_trace_ok,
  'population':{'minted':1000,'active':expected,'royalty_weight':1500 if growth else 1499,
   'source_setup':'999 actual prior Entries, reverse Legendary/Rare/Commun activation order; no raw ledger seeding'},
  'workload':{'royalty_erc20_currencies':64,'royalty_assets_total':65,'royalty_asset_capacity':64,
   'royalty_assets':assets,'royalty_sync_gas':1_000_000,'royalty_ledger':royalty,'mint_helper':helper,
   'seadrop':drop,'actual_empty_code_sender':case['sender'],'actual_workload':workload},
  'physical_portfolio':physical,'enrollment_storage_proof':density}
 trace=trace32['result']if trace32_ok else None;trace_tx=tx
 if padded_trace_ok:trace=padded_trace['result'];trace_tx={**tx,'gas':hex(padded)}
 if trace is None:
  diagnostic_tx={**tx,'gas':hex(DIAGNOSTIC)}
  diagnostic_call=rpc.attempt('eth_call',[diagnostic_tx,anchor]);diagnostic_trace=checked_trace(rpc,diagnostic_tx,anchor,omitted)
  result['diagnostic96']={'execution_only_no_adoption':True,'cap':DIAGNOSTIC,'call':diagnostic_call,
   'trace_succeeded':diagnostic_trace['ok']and not diagnostic_trace['result'].get('error')}
  if diagnostic_call['ok']:assert diagnostic_call['result']=='0x'
  if diagnostic_trace['ok']and not diagnostic_trace['result'].get('error'):
   trace=diagnostic_trace['result'];trace_tx=diagnostic_tx
  result['diagnostic96']['estimate']=rpc.attempt('eth_estimateGas',[diagnostic_tx,anchor])
 assert trace is not None,'No complete financial trace even in explicit test-only diagnostic envelope'
 measured=measure_trace(trace,trace_tx,case,read)
 royalty_proof=royalty_trace(trace,case,royalty,assets,workload)
 residual_proof=residual_trace(trace,case,physical)
 financial=bool(royalty_proof['verified']and residual_proof['verified'])
 # The selected padded trace is checked financially as well as its eth_call.
 # Entry returns no data, so matching a bare success alone cannot prove cohorts.
 result.update(production_trace_measurement=measured,production_trace_measurement_verified=True,
  royalty_trace_verification=royalty_proof,residual_trace_verification=residual_proof,
  trace_gas_used=quantity(trace['gasUsed']),measurement_gas_cap=quantity(trace_tx['gas']),
  measurement_scope='exact-padded32M-profile call'if padded_trace_ok else'32M call without admitted padding'if trace32_ok else'96M TEST-ONLY diagnostic; not profile qualification',
  exact_padded_trace_financial_verified=bool(padded_trace_ok and financial),
  verification_complete=financial,meets_reviewed_execution_profile=bool(admitted and financial))
 price=quantity(rpc.call('eth_gasPrice'));balance=quantity(rpc.call('eth_getBalance',[case['sender'],anchor]))
 result.update(local_gas_price_wei=price,sender_balance_wei=balance,
  fee_qualification='Local execution-only gas price, not complete target L1/L2/provider fee qualification.')
 if admitted:assert balance>=case['value']+padded*price
 return result


def main(argv=None):
 parser=argparse.ArgumentParser(description=__doc__)
 parser.add_argument('stage',help='Exact newly completed owned producerNN stage')
 args=parser.parse_args(argv)
 context=configure(args.stage);cases,forge_hash,source,methods=exports()
 chosen=[case for case in cases if case['label']in CASE_POLICY]
 skipped=[case['label']for case in cases if case['label']in MAINTENANCE_LABELS]
 assert len(chosen)==12 and len(skipped)==8
 out=F/'cold-contributions';out.mkdir(exist_ok=False)
 vectors=[]
 for raw in(b'',b'abc',bytes(range(256))):
  result=native_keccak256(raw);assert result==python_keccak256(raw)
  vectors.append({'input_sha256':sha(raw),'length':len(raw),'keccak256':'0x'+result.hex()})
 start={'schema':1,**context,'checker_sha256':sha(Path(__file__).read_bytes()),
  'measurement_helper_sha256':MEASUREMENT_HELPERS,'normal_binding_sha256':NORMAL_BINDING_SHA256,
  'new_source_methods':methods,'contribution_cases':[case['label']for case in chosen],
  'explicitly_delegated_maintenance_cases':skipped,'public_requests_or_transactions':0,
  'test_only_instrumentation':{'rpc_posts_per_case':1600,'work_seconds_per_case':900,'maximum_reply_bytes':MAX_REPLY,
   'max_parallel_owned_nodes':3,'genesis_diagnostic_gas':DIAGNOSTIC,'not_production_API_limits':True},
  'keccak_crosschecks':vectors}
 (out/'CHECKER_START.json').write_text(json.dumps(start,indent=2)+'\n',encoding='utf-8')
 completed={}
 with ThreadPoolExecutor(max_workers=3)as workers:
  jobs={workers.submit(owned_case,case,out,verify_contribution):case['label']for case in chosen}
  for future in as_completed(jobs):
   result=future.result();completed[jobs[future]]=result
   print(json.dumps({'case':jobs[future],'verification_complete':result['verification_complete'],
    'meets_32m_full20pct':result['meets_reviewed_execution_profile'],'estimated_gas':result.get('estimated_gas'),
    'padded_gas':result.get('padded_gas'),'owned_node_stopped':result['owned_node_stopped'],
    'verification_error':result.get('verification_error')}),flush=True)
 results=[completed[case['label']]for case in chosen]
 stable=all(sha((R/name).read_bytes())==pin for name,pin in MEASUREMENT_HELPERS.items())
 summary={'schema':1,**context,'measurement_helpers_unchanged':stable,'source_methods':methods,
  'new_contribution_scenarios':len(chosen),'completed_scenarios':len(results),
  'explicitly_delegated_maintenance_scenarios':skipped,
  'verification_complete':stable and len(results)==12 and all(row['verification_complete']and row['owned_node_stopped']for row in results),
  'meets_32m_full20pct':[row['profile']['label']for row in results if row['meets_reviewed_execution_profile']],
  'refuses_32m_full20pct':[row['profile']['label']for row in results if not row['meets_reviewed_execution_profile']],
  'results':results,'production_approved':False,'diagnostic96_adopted':False,'unchanged_old_suites_executed':0}
 (out/'SUMMARY.json').write_text(json.dumps(summary,indent=2)+'\n',encoding='utf-8')
 return 0 if summary['verification_complete']else 1
