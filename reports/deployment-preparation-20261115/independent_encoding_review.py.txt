def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def canonical(obj):
    return json.dumps(obj, sort_keys=True, separators=(',',':'), ensure_ascii=True, allow_nan=False).encode('ascii')

def encode(types, values):
    """Independent standard ABI head/tail implementation, checked against Cast."""
    if any(v is None for v in values):
        return None
    head=[]; tail=b''
    for typ,v in zip(types,values):
        if typ in ('string','bytes'):
            payload=v.encode('utf-8') if typ=='string' else v
            head.append((32*len(types)+len(tail)).to_bytes(32,'big'))
            tail+=len(payload).to_bytes(32,'big')+payload+b'\0'*((-len(payload))%32)
        elif typ=='address':
            head.append(bytes.fromhex(v[2:]).rjust(32,b'\0'))
        elif typ=='bytes32':
            head.append(bytes.fromhex(v[2:]))
        elif typ=='uint256':
            head.append(int(v).to_bytes(32,'big'))
        else:
            raise AssertionError(('unexpected constructor type',typ))
    return b''.join(head)+tail
