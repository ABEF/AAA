class RecordedRpc(Rpc):
 def _read_http(self,url,raw):
  global last,stopped
  if stopped:raise RpcError('Public transport was stopped after a denial or deadline')
  if url!=URL or len(rows)>=100 or time.monotonic()-START>=90:
   stopped=True;raise RpcError('Recorded public transport budget exhausted')
  delay=1.5-(time.monotonic()-last)
  if delay>0:time.sleep(delay)
  if time.monotonic()-START>=90:
   stopped=True;raise RpcError('Recorded public transport deadline elapsed')
  request=json.loads(raw)
  if request['method']not in Rpc.READ_METHODS:raise RpcError('Method outside read-only policy')
  last=time.monotonic()
  row={'provider':URL,'observed_at_utc':dt.datetime.now(dt.timezone.utc).isoformat(),
       'request':request,'sequence':len(rows)+1}
  rows.append(row);began=time.monotonic()
  try:
   data=super()._read_http(url,raw)
   filename=f'full-probe-{len(rows):02d}.raw'
   with(HERE/filename).open('xb')as stream:stream.write(data)
   row.update(http_status=200,raw_relative_path=filename,response_sha256=hashlib.sha256(data).hexdigest(),
              response_bytes=len(data),rpc_response=json.loads(data))
   return data
  except Exception as exc:
   row['error_class']=type(exc).__name__
   if isinstance(exc,urllib.error.HTTPError):
    row['http_status']=exc.code
    retry=exc.headers.get('Retry-After')if exc.headers is not None else None
    if isinstance(retry,str)and len(retry)<=128:row['retry_after']=retry
    if exc.code in(401,403,429):stopped=True
   raise
  finally:row['duration_ms']=round((time.monotonic()-began)*1000,3)
